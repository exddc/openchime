function(oc_read_first_line path out_var)
  set(_value "")
  if(EXISTS "${path}")
    file(STRINGS "${path}" _lines LIMIT_COUNT 1)
    if(_lines)
      list(GET _lines 0 _value)
      string(STRIP "${_value}" _value)
    endif()
  endif()
  set(${out_var} "${_value}" PARENT_SCOPE)
endfunction()

function(oc_genex_targets expr out_var)
  set(_found "")
  if(NOT "${expr}" STREQUAL "")
    string(REGEX MATCHALL "[A-Za-z_][A-Za-z0-9_.:-]*" _tokens "${expr}")
    foreach(_tok IN LISTS _tokens)
      if(TARGET "${_tok}")
        list(APPEND _found "${_tok}")
      endif()
    endforeach()
  endif()
  set(${out_var} "${_found}" PARENT_SCOPE)
endfunction()

function(oc_collect_link_closure target out_targets out_genex)
  if(NOT TARGET "${target}")
    message(FATAL_ERROR "missing target ${target}")
  endif()
  set(_stack "${target}")
  set(_seen "")
  set(_targets "")
  set(_genex "")
  while(_stack)
    list(POP_FRONT _stack _cur)
    if(TARGET "${_cur}")
      get_target_property(_aliased "${_cur}" ALIASED_TARGET)
      if(_aliased)
        set(_cur "${_aliased}")
      endif()
    endif()
    if("${_cur}" IN_LIST _seen)
      continue()
    endif()
    list(APPEND _seen "${_cur}")
    if(NOT TARGET "${_cur}")
      continue()
    endif()
    list(APPEND _targets "${_cur}")
    set(_defs "")
    get_target_property(_cd "${_cur}" COMPILE_DEFINITIONS)
    get_target_property(_icd "${_cur}" INTERFACE_COMPILE_DEFINITIONS)
    if(_cd)
      list(APPEND _defs ${_cd})
    endif()
    if(_icd)
      list(APPEND _defs ${_icd})
    endif()
    foreach(_def IN LISTS _defs)
      if(_def MATCHES "\\$<")
        oc_genex_targets("${_def}" _from_def)
        if(_from_def)
          list(APPEND _stack ${_from_def})
        endif()
      endif()
    endforeach()
    get_target_property(_type "${_cur}" TYPE)
    set(_libs "")
    if(_type STREQUAL "INTERFACE_LIBRARY")
      get_target_property(_libs "${_cur}" INTERFACE_LINK_LIBRARIES)
    else()
      get_target_property(_direct "${_cur}" LINK_LIBRARIES)
      get_target_property(_iface "${_cur}" INTERFACE_LINK_LIBRARIES)
      if(_direct)
        list(APPEND _libs ${_direct})
      endif()
      if(_iface)
        list(APPEND _libs ${_iface})
      endif()
    endif()
    foreach(_lib IN LISTS _libs)
      if(NOT _lib)
        continue()
      endif()
      if(_lib MATCHES "\\$<")
        list(APPEND _genex "${_lib}")
        oc_genex_targets("${_lib}" _from_lib)
        if(_from_lib)
          list(APPEND _stack ${_from_lib})
        endif()
        while(_lib MATCHES "^\\$<[A-Za-z0-9_]+:([^,>]+)>$")
          set(_lib "${CMAKE_MATCH_1}")
        endwhile()
        if(_lib MATCHES "\\$<")
          continue()
        endif()
      endif()
      list(APPEND _stack "${_lib}")
    endforeach()
  endwhile()
  set(${out_targets} "${_targets}" PARENT_SCOPE)
  set(${out_genex} "${_genex}" PARENT_SCOPE)
endfunction()

function(oc_assert_no_product_payload root via text seen)
  if("${text}" STREQUAL "")
    return()
  endif()
  if(text MATCHES "(^|[^A-Za-z0-9_])(CHIME_|OPENCHIME_)")
    message(FATAL_ERROR
      "${root} must not compile with product definition ${text} (via ${via})")
  endif()
  if(text MATCHES "(^|[^A-Za-z0-9_])oc_build_version([^A-Za-z0-9_]|$)")
    message(FATAL_ERROR
      "${root} must not depend on oc_build_version via ${text}")
  endif()
  if(NOT text MATCHES "TARGET_PROPERTY")
    return()
  endif()
  string(REGEX MATCHALL "\\$<TARGET_PROPERTY:[^$<>]+>" _refs "${text}")
  if(NOT _refs)
    message(FATAL_ERROR
      "${root} must not compile with product definition ${text} (via ${via})")
  endif()
  foreach(_ref IN LISTS _refs)
    string(REGEX REPLACE "^\\$<TARGET_PROPERTY:" "" _body "${_ref}")
    string(REGEX REPLACE ">$" "" _body "${_body}")
    if(_body MATCHES "^([^,]+),(.+)$")
      set(_tgt "${CMAKE_MATCH_1}")
      set(_prop "${CMAKE_MATCH_2}")
    else()
      set(_tgt "${via}")
      set(_prop "${_body}")
    endif()
    if(NOT TARGET "${_tgt}")
      continue()
    endif()
    get_target_property(_aliased "${_tgt}" ALIASED_TARGET)
    if(_aliased)
      set(_tgt "${_aliased}")
    endif()
    set(_key "${_tgt}|${_prop}")
    if("${_key}" IN_LIST seen)
      continue()
    endif()
    set(_seen "${seen}")
    list(APPEND _seen "${_key}")
    get_property(_val TARGET "${_tgt}" PROPERTY "${_prop}")
    oc_assert_no_product_payload("${root}" "${via}" "${_val}" "${_seen}")
  endforeach()
endfunction()

function(oc_assert_platform_boundary target)
  oc_collect_link_closure("${target}" _targets _genex)
  foreach(_cur IN LISTS _targets)
    if(_cur MATCHES "^chime")
      message(FATAL_ERROR
        "${target} must not link product target ${_cur}")
    endif()
    if(_cur STREQUAL "oc_build_version")
      message(FATAL_ERROR
        "${target} must not depend on oc_build_version")
    endif()
    set(_defs "")
    get_target_property(_cd "${_cur}" COMPILE_DEFINITIONS)
    get_target_property(_icd "${_cur}" INTERFACE_COMPILE_DEFINITIONS)
    if(_cd)
      list(APPEND _defs ${_cd})
    endif()
    if(_icd)
      list(APPEND _defs ${_icd})
    endif()
    foreach(_def IN LISTS _defs)
      oc_assert_no_product_payload("${target}" "${_cur}" "${_def}" "")
    endforeach()
  endforeach()
  foreach(_lib IN LISTS _genex)
    if(_lib MATCHES "(^|[^A-Za-z0-9_])chime[A-Za-z0-9_]*")
      message(FATAL_ERROR
        "${target} must not link product target via ${_lib}")
    endif()
    if(_lib MATCHES "oc_build_version")
      message(FATAL_ERROR
        "${target} must not depend on oc_build_version via ${_lib}")
    endif()
  endforeach()
endfunction()
